<script>
  // gera pétalas caindo continuamente
  const container = document.getElementById('petalas');
  const emojis = ['🌸','🌺','💗','🌷'];
  const totalPetalas = 22;

  for (let i = 0; i < totalPetalas; i++) {
    const p = document.createElement('span');
    p.className = 'petala';
    p.textContent = emojis[Math.floor(Math.random() * emojis.length)];
    const left = Math.random() * 100;
    const duracao = 8 + Math.random() * 10;
    const atraso = Math.random() * 12;
    const drift = (Math.random() * 120 - 60) + 'px';
    p.style.left = left + 'vw';
    p.style.fontSize = (14 + Math.random() * 14) + 'px';
    p.style.setProperty('--drift', drift);
    p.style.animationDuration = duracao + 's';
    p.style.animationDelay = atraso + 's';
    container.appendChild(p);
  }
